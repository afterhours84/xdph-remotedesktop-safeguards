#!/bin/bash
# xdph-rd-on-lock: run when xdph closes remote sessions because the screen locked.
# xdph already double-forks this hook. We only SIGKILL rustdesk here and arm a
# one-shot unlock waiter; rustdesk is NOT restarted until unlock.

LOG="${HOME}/.local/share/xdph-rd/lock-hook.log"
mkdir -p "$(dirname "$LOG")"
ts() { date '+%Y-%m-%d %H:%M:%S %z'; }

echo "$(ts) xdph-rd-on-lock: invoked (screen locked); hard-killing rustdesk (no restart until unlock)" >>"$LOG"

pkill -KILL -x rustdesk 2>/dev/null || true
pkill -KILL -f '/usr/bin/rustdesk' 2>/dev/null || true
pkill -KILL -f '/usr/lib/rustdesk/rustdesk' 2>/dev/null || true
sleep 0.3
if pgrep -x rustdesk >/dev/null 2>&1; then
  echo "$(ts) xdph-rd-on-lock: WARN rustdesk still present after KILL: $(pgrep -ax rustdesk | tr '\n' '; ')" >>"$LOG"
else
  echo "$(ts) xdph-rd-on-lock: rustdesk gone after KILL" >>"$LOG"
fi

# Arm unlock waiter (exactly one via flock). Detached so this hook returns now.
WAIT="${HOME}/.local/bin/xdph-rd-wait-unlock"
LOCKFILE="${XDG_RUNTIME_DIR:-/tmp}/xdph-rd-wait-unlock.lock"
if [ -x "$WAIT" ]; then
  setsid -f bash -c "flock -n '$LOCKFILE' '$WAIT' || echo \"\$(date '+%Y-%m-%d %H:%M:%S %z') xdph-rd-on-lock: unlock waiter already running; leave it\" >>'$LOG'" >/dev/null 2>&1
  echo "$(ts) xdph-rd-on-lock: unlock waiter spawn attempted" >>"$LOG"
else
  echo "$(ts) xdph-rd-on-lock: WARN missing $WAIT" >>"$LOG"
fi

echo "$(ts) xdph-rd-on-lock: done" >>"$LOG"
exit 0

#!/bin/bash
# xdph-rd-wait-unlock: poll until the compositor session lock is gone, then
# run xdph-rd-on-unlock. Intended to be run under flock so only one waiter
# exists. Omarchy has no unlock hook and does not use hypridle; this uses the
# same helper Omarchy uses (omarchy-hyprland-session-locked).

LOG="${HOME}/.local/share/xdph-rd/lock-hook.log"
mkdir -p "$(dirname "$LOG")"
ts() { date '+%Y-%m-%d %H:%M:%S %z'; }

echo "$(ts) xdph-rd-wait-unlock: waiting for unlock (pid $$)" >>"$LOG"

# Exit 0 = locked. Exit 1 = unlocked. Exit 2 = undetermined (treat as keep waiting
# while we know a lock just happened; only break on clear unlock).
while true; do
  if omarchy-hyprland-session-locked; then
    # locked
    sleep 1
    continue
  fi
  ec=$?
  if [ "$ec" -eq 1 ]; then
    echo "$(ts) xdph-rd-wait-unlock: unlocked (omarchy-hyprland-session-locked exit 1)" >>"$LOG"
    break
  fi
  # undetermined (2) or other: confirm via omarchy-shell lock status when available
  status=$(omarchy-shell lock status 2>/dev/null || true)
  if printf '%s' "$status" | grep -q '"locked":false' \
     && printf '%s' "$status" | grep -q '"sessionLocked":false'; then
    echo "$(ts) xdph-rd-wait-unlock: unlocked (shell status: $status)" >>"$LOG"
    break
  fi
  sleep 1
done

ONUNLOCK="${HOME}/.local/bin/xdph-rd-on-unlock"
if [ -x "$ONUNLOCK" ]; then
  exec "$ONUNLOCK"
fi
echo "$(ts) xdph-rd-wait-unlock: WARN missing $ONUNLOCK" >>"$LOG"
exit 1

#!/bin/bash
# xdph-rd-on-unlock: start RustDesk after the session unlocks (idempotent).
# Called by xdph-rd-wait-unlock once omarchy-hyprland-session-locked reports unlocked.

LOG="${HOME}/.local/share/xdph-rd/lock-hook.log"
mkdir -p "$(dirname "$LOG")"
ts() { date '+%Y-%m-%d %H:%M:%S %z'; }

echo "$(ts) xdph-rd-on-unlock: invoked" >>"$LOG"

if pgrep -x rustdesk >/dev/null 2>&1; then
  echo "$(ts) xdph-rd-on-unlock: rustdesk already running; skip start: $(pgrep -ax rustdesk | tr '\n' '; ')" >>"$LOG"
  exit 0
fi

echo "$(ts) xdph-rd-on-unlock: starting /usr/bin/rustdesk via setsid" >>"$LOG"
setsid -f /usr/bin/rustdesk
sleep 1
echo "$(ts) xdph-rd-on-unlock: after start: $(pgrep -ax rustdesk 2>/dev/null | tr '\n' '; ' || echo none)" >>"$LOG"
exit 0
